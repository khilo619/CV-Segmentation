\section{Executive Summary \& Problem Definition}

\subsection{Business Context to Technical Translation}
Enterprise Applicant Tracking Systems (ATS) including Workday, Greenhouse, Taleo, and Bullhorn routinely merge hundreds of multi-page candidate resumes into a single monolithic PDF stream (typically 200 to 1,000 pages). Feeding these unsegmented bulk files directly into downstream automated parsers generates catastrophic \textbf{"PII chimeras"} (fabricating hybrid identities by merging distinct candidates) and triggers memory exhaustion or payload timeouts. 

This engineering backlog defines the complete, prioritized roadmap for building an automated, high-throughput, on-premise engine to detect candidate transitions and perform zero-copy lossless slicing.

\subsection{Core Architectural Tenets}
\begin{itemize}[leftmargin=1.5em, itemsep=3pt]
    \item \textbf{CPU-First Low-Latency Budget:} Sub-15 millisecond per-page inference budget on commodity CPU without requiring GPU acceleration or heavy Vision Transformers (ViT).
    \item \textbf{Lossless Zero-Copy Slicing:} Direct byte-level PDF cross-reference table manipulation via \texttt{pikepdf} to guarantee 100\% vector, font, and attachment fidelity.
    \item \textbf{Tiered Cascade Decision Engine:} Fast heuristic gating (Tier 1) $\to$ 36-dimensional pairwise LightGBM (Tier 2) $\to$ Global Viterbi sequence optimization (Tier 3).
\end{itemize}

\subsection{Definition of Ready (DoR) \& Definition of Done (DoD)}
\begin{itemize}[leftmargin=1.5em, itemsep=3pt]
    \item \textbf{Definition of Ready (DoR):} User story includes clear business justification, testable technical acceptance criteria, architectural tier mapping, story points (Fibonacci), and assigned engineer ownership.
    \item \textbf{Definition of Done (DoD):} Implemented with full Python 3.11 static typing (\texttt{mypy src/ --strict}), covered by unit and integration tests (\texttt{pytest}), code coverage $\ge 70\%$ (progressing to $\ge 85\%$), verified with \texttt{ruff}, peer reviewed, and merged into \texttt{main}.
\end{itemize}

\section{Detailed User Stories \& Technical Acceptance Criteria}

\subsection{EPIC-1: High-Throughput Digital Ingestion \& Conditional OCR Triage}
\textit{Architectural Domain: Ingestion Tier --- Lead: Engineer 1 (Repo Manager / IO Lead) --- Total Points: 6}

\vspace{0.5em}

\begin{storybox}{Story 1.1: Native PyMuPDF Digital Text \& Geometry Extraction (Points: 3 $|$ Priority: P0)}
\textbf{As an} ML pipeline, \\
\textbf{I want} structured text spans, bounding boxes, and font sizes extracted from digital PDFs in $<4\text{ ms}$ per page, \\
\textbf{So that} downstream models receive rich layout geometry without paying rasterization or OCR latency penalties.

\vspace{0.3em}
\textbf{Technical Acceptance Criteria:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item \texttt{PDFStreamParser.parse\_pdf()} returns an ordered sequence of \texttt{PagePayload} dataclasses.
    \item Extracts text spans with $(x_0, y_0, x_1, y_1)$ baseline bounding boxes, font name, and font point size.
    \item Correctly calculates printable character count and detects embedded image objects.
    \item Handles missing files and corrupt streams with clean exceptions.
\end{itemize}
\textbf{Assignee:} Engineer 1 (Repo Manager / IO Lead)
\end{storybox}

\begin{storybox}{Story 1.2: Conditional OCR Triage Subsystem (Points: 3 $|$ Priority: P1)}
\textbf{As a} system administrator, \\
\textbf{I want} OCR to be triggered strictly on scanned or flatbed resume pages, \\
\textbf{So that} processing digital PDFs remains under 15 milliseconds per page.

\vspace{0.3em}
\textbf{Technical Acceptance Criteria:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item Evaluates printable character density threshold ($\text{chars} < 80$ AND has embedded raster image).
    \item Bypasses OCR on born-digital pages ($\ge 95\%$ of standard recruitment streams).
    \item Invokes \texttt{PaddleOCR} / \texttt{Tesseract} only on confirmed scanned pages to synthesize text spans.
    \item Preserves identical \texttt{PagePayload} schema output so downstream tiers are format-agnostic.
\end{itemize}
\textbf{Assignee:} Engineer 1
\end{storybox}

\begin{storybox}{Story 2.2: Anti-Leakage Synthetic Stream Generator (Points: 5 $|$ Priority: P0)}

\end{itemize}

\subsection{EPIC-3: Production Heuristic Gating Engine (Tier 1 Rules --- Part I)}
\textit{Architectural Domain: Deterministic Gating --- Lead: Engineer 2 (Features / Rules Lead) --- Subtotal: 5 Points}

\vspace{0.5em}

\begin{storybox}{Story 3.1: Header PII Density Signal (Points: 3 $|$ Priority: P0)}
\textbf{As a} boundary detector, \\
\textbf{I want} to identify candidate contact credentials in page headers, \\
\textbf{So that} new candidate start pages are recognized with high deterministic confidence.

\vspace{0.3em}
\textbf{Technical Acceptance Criteria:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item Scans top 35\% vertical coordinate zone of each page for email, international phone, LinkedIn, and GitHub patterns.
    \item Computes weighted score $S_{\text{PII}}$. When $S_{\text{PII}} \ge 3.0$ and preceding page has low PII, emits boundary probability $P \ge 0.98$.
    \item Benchmarked execution speed: $<1\text{ ms}$ per page.
\end{itemize}
\textbf{Assignee:} Engineer 2 (Heuristics \& Feature Lead)
\end{storybox}

\begin{storybox}{Story 3.2: Typography Hierarchy \& Dominance Ratio Analyzer (Points: 2 $|$ Priority: P1)}
\textbf{As a} boundary detector, \\
\textbf{I want} to measure font dominance ratios across page headers, \\
\textbf{So that} large typographic name titles reliably indicate fresh candidate documents.

\vspace{0.3em}
\textbf{Technical Acceptance Criteria:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item Computes Dominance Ratio $R_{\text{font}} = \max(\text{top } 25\%) / \text{median}(\text{all spans})$.
    \item Flags transitions where $R_{\text{font}}(P_{i+1}) \ge 1.8$ and $R_{\text{font}}(P_i) < 1.3$ as candidate starts.
    \item Emits deterministic start signal when paired with high header whitespace fraction.
\end{itemize}
\textbf{Assignee:} Engineer 2
\end{storybox}

\subsection{EPIC-3: Production Heuristic Gating Engine (Tier 1 Rules --- Part II)}
\textit{Architectural Domain: Deterministic Gating --- Lead: Engineer 2 (Features / Rules Lead) --- Subtotal: 6 Points}

\vspace{0.5em}

\begin{storybox}{Story 3.3: Explicit Header/Footer Pagination Tracking (Points: 3 $|$ Priority: P0)}
\textbf{As a} boundary detector, \\
\textbf{I want} to parse explicit pagination patterns in header and footer margins, \\
\textbf{So that} explicit sequence markers provide deterministic ground-truth boundary decisions.

\vspace{0.3em}
\textbf{Technical Acceptance Criteria:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item Matches \texttt{"Page X of Y"} and solitary margin numbers in top/bottom 10\% zones.
    \item Emits boundary ($y_i = 1$) when next page is \texttt{"Page 1"} or current page is \texttt{"Page k of k"}.
    \item Emits continuation ($y_i = 0$) when current is $k$ and next is $k+1$.
    \item Suppresses noisy margin numbers occurring inside page body text blocks.
\end{itemize}
\textbf{Assignee:} Engineer 2
\end{storybox}

\begin{storybox}{Story 3.4: Cross-Page Syntactic Sentence Fracture Analyzer (Points: 3 $|$ Priority: P1)}
\textbf{As a} boundary detector, \\
\textbf{I want} to detect cross-page grammatical continuations, \\
\textbf{So that} mid-sentence physical page splits are never erroneously broken into new candidates.

\vspace{0.3em}
\textbf{Technical Acceptance Criteria:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item Detects trailing hyphens on $P_i$ followed by lowercase token on $P_{i+1}$ (continuation score $= 0.99$).
    \item Detects non-terminal punctuation with trailing coordinating conjunctions (\texttt{and}, \texttt{with}, \texttt{or}, \texttt{to}) followed by lowercase text (continuation score $= 0.95$).
    \item Short-circuits downstream ML when continuation certainty is high.
\end{itemize}
\textbf{Assignee:} Engineer 2
\end{storybox}

\begin{storybox}{Story 4.1: Pairwise Tabular Feature Extractor (Points: 5 $|$ Priority: P0)}

\vspace{0.5em}

\begin{storybox}{Story 5.1: Hidden Markov Model Transition Priors (Points: 3 $|$ Priority: P0)}
\textbf{As a} sequence optimizer, \\
\textbf{I need} transition matrix $\mathbf{A}$ over states $\{\text{START}, \text{PAGE\_2}, \text{PAGE\_3}, \text{PAGE\_4+}\}$, \\

\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item Implements empirical resume length transition matrix $\mathbf{A}$ derived from recruitment statistics.
    \item Structurally enforces $P(\text{START} \to \text{PAGE\_3}) = 0.00$ and eliminates orphaned continuation pages.
    \item Allows parameter override for executive search streams with longer average CV lengths.
\end{itemize}
\textbf{Assignee:} Engineer 3

\begin{storybox}{Story 5.2: Viterbi Trellis Dynamic Programming (Points: 3 $|$ Priority: P0)}
\textbf{As a} pipeline orchestrator, \\
\textbf{I want} forward log-space trellis induction and backtracking over the page sequence, \\
\textbf{So that} local classifier probabilities are globally smoothed into valid candidate documents.

\vspace{0.3em}
\textbf{Technical Acceptance Criteria:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item \texttt{ViterbiSequenceOptimizer.decode()} computes globally optimal path $\mathbf{S}^\ast$.
    \item Decodes a 500-page document stream in $<1\text{ millisecond}$ on CPU.
    \item Emits 1-indexed split cut points matching slicing engine requirements.
\end{itemize}
\textbf{Assignee:} Engineer 3 \& Engineer 1
